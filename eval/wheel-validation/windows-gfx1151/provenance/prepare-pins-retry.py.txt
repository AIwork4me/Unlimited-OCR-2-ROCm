import subprocess,pathlib,json,datetime
root=pathlib.Path('C:/b/pytorch'); records=[]
lines=subprocess.check_output(['git','-C',str(root),'ls-tree','-r','HEAD'],text=True).splitlines()
with open('C:/b/raw/submodule-pinned-retry.txt','w',encoding='utf-8') as log:
    for line in lines:
        if not line.startswith('160000 '):continue
        meta,name=line.split('\t');sha=meta.split()[2];path=root/name
        if not (path/'.git').exists():continue
        probe=subprocess.run(['git','-C',str(path),'rev-parse','--verify','HEAD'],stdout=subprocess.DEVNULL,stderr=subprocess.DEVNULL)
        if probe.returncode==0:continue
        print('Pinned source retry:',name,sha,flush=True)
        cmd=['git','-C',str(path),'-c','http.version=HTTP/1.1','-c','http.sslBackend=openssl','-c','http.lowSpeedLimit=1024','-c','http.lowSpeedTime=20','fetch','--depth','1','origin',sha]
        record={'path':name,'sha':sha,'command':cmd,'start_utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),'timeout_seconds':120}
        p=subprocess.Popen(cmd,stdout=log,stderr=subprocess.STDOUT)
        try:record['exit_code']=p.wait(timeout=120)
        except subprocess.TimeoutExpired:
            subprocess.run(['taskkill.exe','/PID',str(p.pid),'/T','/F'],stdout=log,stderr=subprocess.STDOUT)
            record.update(exit_code=124,actual_terminated_process_exit=p.wait())
        record['end_utc']=datetime.datetime.now(datetime.timezone.utc).isoformat()
        records.append(record);pathlib.Path('C:/b/raw/submodule-pinned-retry.json').write_text(json.dumps({'records':records},indent=2))
        if record['exit_code']:raise SystemExit(record['exit_code'])
        r=subprocess.run(['git','-C',str(path),'-c','http.version=HTTP/1.1','-c','http.sslBackend=openssl','checkout','--detach',sha],stdout=log,stderr=subprocess.STDOUT)
        if r.returncode:raise SystemExit(r.returncode)
print('Pinned preparation complete',flush=True)
