base = pathlib.Path('.local/pdf-pipeline-rca')
def read(path):
    return json.loads(pathlib.Path(path).read_text(encoding='utf-8'))
def sha(path):
    digest = hashlib.sha256()
    with pathlib.Path(path).open('rb') as stream:
        for block in iter(lambda: stream.read(8 * 1024 * 1024), b''):
            digest.update(block)
    return digest.hexdigest()
def save(path, value):
    pathlib.Path(path).write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n', encoding='utf-8')

render = read(base / 'render-manifest.json')

lock = read_lock()
expected = {**lock['files'], **lock['weights']}
snapshot = pathlib.Path.home() / '.cache/unlimited-ocr2-phase1/models/models--baidu--Unlimited-OCR-2/snapshots' / lock['model_revision']

def grams(text):
    text = normalize(text).casefold()
    return {text[i:i+5] for i in range(max(0, len(text)-4))}
refs = read(base / 'pdf-text-layer-reference.json')
reference = [grams(r['text']) for r in refs]
rows = []
for i, part in enumerate(segments, 1):
    g = grams(part)
    scores = [2 * len(g & r) / max(1, len(g) + len(r)) for r in reference]
    best = max(range(len(scores)), key=lambda j: scores[j])
    rows.append({'output_segment': i, 'best_matching_PDF_page': best + 1, 'text_fingerprint_dice': scores[best], 'same_index_score': scores[i-1], 'next_best_score': max(s for j, s in enumerate(scores) if j != best)})
order = {'purpose': 'Diagnostic normalized 5-character-gram set Dice against the PDF text layer; not CER/WER, manually verified ground truth or an OCR acceptance threshold. Text layer was never model input.', 'threshold': None, 'matching_page_indices': sum(r['output_segment'] == r['best_matching_PDF_page'] for r in rows), 'rows': rows}
save(base / 'full-order-diagnostic.json', order)
